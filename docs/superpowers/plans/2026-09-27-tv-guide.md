# TV Guide (Now/Next) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Show "On now" (with progress) and "Up next" in the player for channels that have i.mjh.nz schedule data.

**Architecture:** A build-time script maps iptv-org channel feeds to i.mjh.nz schedule files (`public/guide-map.json`). When the player opens a mapped channel, a Web Worker downloads that channel's XMLTV file, decompresses it, and parses the 13-hour window into plain programme lists. A React hook caches one result per file for the session, and a `GuidePanel` renders it under the video.

**Tech Stack:** React 19, TypeScript 5.9, Vite 8 (module workers via `new URL(..., import.meta.url)`), Tailwind 4, Vitest (new, dev-only), browser `DecompressionStream`.

**Spec:** `docs/superpowers/specs/2026-09-27-tv-guide-design.md`

## Global Constraints

- No backend. Schedule data is fetched from `https://raw.githubusercontent.com/matthuisman/i.mjh.nz/refs/heads/master` (not `i.mjh.nz`, whose redirect fails CORS in Chrome).
- Guide window: now − 1 h to now + 12 h.
- Up next shows 4 programmes. Descriptions are trimmed to 280 characters.
- Channels without a mapping render no guide UI at all.
- A failed schedule file is not retried within the session.
- Copy is sentence case: "On now", "Up next", "Guide unavailable", footer "Guide data from i.mjh.nz."
- Visual tokens come from `src/styles/index.css` `@theme` (ink/panel/raised/line/paper/dim/amber/onair; font-display/sans/mono). Add no new colors or fonts.
- Match the existing code style: 2-space indentation, single quotes, semicolons, named exports, comments only where the reason is not obvious.

Create `src/app/guide/parseXmltv.ts`:

```ts
export interface Programme {
  /** Epoch milliseconds. */
  start: number;
  stop: number;
  title: string;
  subtitle?: string;
  description?: string;
}

const MAX_DESCRIPTION = 280;
const TIME = /^(\d{4})(\d{2})(\d{2})(\d{2})(\d{2})(\d{2})(?:\s*([+-])(\d{2})(\d{2}))?$/;
const NAMED_ENTITIES: Record<string, string> = { amp: '&', lt: '<', gt: '>', quot: '"', apos: "'" };

/** "YYYYMMDDhhmmss ±hhmm" → epoch ms. No offset is read as UTC. NaN when malformed. */
export function parseXmltvTime(value: string): number {
  const match = TIME.exec(value.trim());
  if (!match) return Number.NaN;
  const [, year, month, day, hour, minute, second, sign, offsetHours, offsetMinutes] = match;
  const utc = Date.UTC(+year, +month - 1, +day, +hour, +minute, +second);
  if (!sign) return utc;
  const offset = (Number(offsetHours) * 60 + Number(offsetMinutes)) * 60_000;
  return sign === '+' ? utc - offset : utc + offset;
}

export function decodeEntities(text: string): string {
  return text.replace(/&(#x[0-9a-f]+|#\d+|[a-z]+);/gi, (entity, body: string) => {
    if (body[0] !== '#') return NAMED_ENTITIES[body.toLowerCase()] ?? entity;
    const code = body[1] === 'x' || body[1] === 'X' ? Number.parseInt(body.slice(2), 16) : Number.parseInt(body.slice(1), 10);
    return Number.isFinite(code) && code <= 0x10ffff ? String.fromCodePoint(code) : entity;
  });
}

function readAttributes(tag: string): Record<string, string> {
  const attributes: Record<string, string> = {};
  for (const match of tag.matchAll(/([\w:-]+)\s*=\s*(?:"([^"]*)"|'([^']*)')/g)) {
    attributes[match[1]] = decodeEntities(match[2] ?? match[3]);
  }
  return attributes;
}

function childText(body: string, name: string): string | undefined {
  const match = new RegExp(`<${name}(?:\\s[^>]*)?>([\\s\\S]*?)</${name}>`).exec(body);
  if (!match) return undefined;
  const text = decodeEntities(match[1].replace(/<!\[CDATA\[([\s\S]*?)\]\]>/g, '$1')).trim();
  return text || undefined;
}

/**
 * Scans XMLTV text (a string scanner: workers have no DOMParser) and returns programmes per channel id
 * that overlap [windowStart, windowEnd], sorted by start.
 */
export function parseXmltv(xml: string, windowStart: number, windowEnd: number): Map<string, Programme[]> {
  const byChannel = new Map<string, Programme[]>();

  for (const match of xml.matchAll(/<programme\b([^>]*)>([\s\S]*?)<\/programme>/g)) {
    const attributes = readAttributes(match[1]);
    const channel = attributes.channel;
    const start = parseXmltvTime(attributes.start ?? '');
    if (!channel || Number.isNaN(start)) continue;
    const stop = attributes.stop ? parseXmltvTime(attributes.stop) : Number.NaN;
    // Cheap early exit for the bulk of a large file. Programmes without a stop are kept until their
    // neighbours are known.
    if (!Number.isNaN(stop) && (stop <= windowStart || start >= windowEnd)) continue;

    const title = childText(match[2], 'title');
    if (!title) continue;
    const programme: Programme = { start, stop, title };
    const subtitle = childText(match[2], 'sub-title');
    if (subtitle) programme.subtitle = subtitle;
    const description = childText(match[2], 'desc');
    if (description) {
      programme.description =
        description.length > MAX_DESCRIPTION ? `${description.slice(0, MAX_DESCRIPTION - 1).trimEnd()}…` : description;
    }

    const list = byChannel.get(channel);
    if (list) list.push(programme);
    else byChannel.set(channel, [programme]);
  }

  const result = new Map<string, Programme[]>();
  for (const [channel, list] of byChannel) {
    list.sort((a, b) => a.start - b.start);
    const kept: Programme[] = [];
    list.forEach((programme, index) => {
      if (Number.isNaN(programme.stop)) {
        const next = list[index + 1];
        if (!next) return;
        programme.stop = next.start;
      }
      if (programme.stop > programme.start && programme.stop > windowStart && programme.start < windowEnd) {
        kept.push(programme);
      }
    });
    if (kept.length) result.set(channel, kept);
  }
  return result;
}
```

import { parseXmltv, type Programme } from './parseXmltv';

/**
 * i.mjh.nz schedules, fetched from where i.mjh.nz redirects to. The i.mjh.nz URL itself fails CORS in
 * browsers because the intermediate github.com redirect sends an empty Access-Control-Allow-Origin.
 */
export const GUIDE_BASE = 'https://raw.githubusercontent.com/matthuisman/i.mjh.nz/refs/heads/master';

const PAST_MS = 60 * 60 * 1000;
const AHEAD_MS = 12 * 60 * 60 * 1000;

async function fetchXml(file: string, fetchImpl: typeof fetch): Promise<string> {
  // The .gz file is served as raw bytes (no Content-Encoding), so it's decompressed here.
  if (typeof DecompressionStream !== 'undefined') {
    try {
      const response = await fetchImpl(`${GUIDE_BASE}/${file}.xml.gz`);
      if (response.ok && response.body) {
        return await new Response(response.body.pipeThrough(new DecompressionStream('gzip'))).text();
      }
    } catch {
      // Fall through to the uncompressed file.
    }
  }
  const response = await fetchImpl(`${GUIDE_BASE}/${file}.xml`);
  if (!response.ok) throw new Error(`Guide file ${file}: HTTP ${response.status}`);
  return response.text();
}

/** Downloads one schedule file and returns programmes per XMLTV channel id, from 1 h ago to 12 h ahead. */
export async function loadSchedule(
  file: string,
  now = Date.now(),
  fetchImpl: typeof fetch = fetch,
): Promise<Record<string, Programme[]>> {
  const xml = await fetchXml(file, fetchImpl);
  return Object.fromEntries(parseXmltv(xml, now - PAST_MS, now + AHEAD_MS));
}
```

import { loadSchedule } from './loadSchedule';
import type { Programme } from './parseXmltv';

export type GuideRequest = { file: string };
export type GuideResponse =
  | { file: string; ok: true; channels: Record<string, Programme[]> }
  | { file: string; ok: false; error: string };

// Stateless: the session cache lives with the caller (useGuide), which decides when to ask.
self.onmessage = async (event: MessageEvent<GuideRequest>) => {
  const { file } = event.data;
  let response: GuideResponse;
  try {
    response = { file, ok: true, channels: await loadSchedule(file) };
  } catch (error) {
    response = { file, ok: false, error: error instanceof Error ? error.message : String(error) };
  }
  self.postMessage(response);
};
```

// Builds public/guide-map.json: iptv-org channel feed ("<channel>@<feed>", the app's Stream.id)
// → i.mjh.nz schedule "<file>#<xmltv channel id>". Run with `npm run guide-map` when coverage should refresh.
import { writeFile } from 'node:fs/promises';

const SOURCE = 'https://iptv-org.github.io/api/guides.json';
const OUTPUT = new URL('../public/guide-map.json', import.meta.url);

const response = await fetch(SOURCE);
if (!response.ok) {
  console.error(`Couldn't fetch ${SOURCE}: HTTP ${response.status}`);
  process.exit(1);
}
const guides = await response.json();

const map = new Map();
for (const guide of guides) {
  if (guide.site !== 'i.mjh.nz' || !guide.channel || !guide.site_id?.includes('#')) continue;
  const key = `${guide.channel}@${guide.feed ?? ''}`;
  if (!map.has(key)) map.set(key, guide.site_id);
}

if (map.size === 0) {
  console.error('No i.mjh.nz entries found; not writing an empty map.');
  process.exit(1);
}

// Sorted keys keep diffs small when the map is regenerated.
const json = JSON.stringify(Object.fromEntries([...map].sort(([a], [b]) => a.localeCompare(b))));
await writeFile(OUTPUT, `${json}\n`);
console.log(`Wrote ${map.size} entries (${Math.round(json.length / 1024)} KB) to public/guide-map.json`);
```

import { useEffect, useState } from 'react';
import type { Stream } from '../types';
import type { Programme } from './parseXmltv';
import type { GuideRequest, GuideResponse } from './guide.worker';

export type GuideState =
  | { status: 'none' }
  | { status: 'loading' }
  | { status: 'ready'; programmes: Programme[] }
  | { status: 'unavailable' };

type GuideMap = Record<string, string>;

export function guideEntry(map: GuideMap, streamId: string): { file: string; channelId: string } | null {
  const value = map[streamId];
  const hash = value?.lastIndexOf('#') ?? -1;
  if (!value || hash <= 0) return null;
  return { file: value.slice(0, hash), channelId: value.slice(hash + 1) };
}

// Session-wide state shared by every player instance.
let guideMap: GuideMap | null = null;
let guideMapPromise: Promise<GuideMap> | null = null;
let worker: Worker | null = null;
/** One promise per schedule file. Failures stay cached so a broken file isn't refetched on every channel change. */
const schedules = new Map<string, Promise<Record<string, Programme[]>>>();

function loadGuideMap(): Promise<GuideMap> {
  guideMapPromise ??= fetch('/guide-map.json')
    .then((response) => {
      if (!response.ok) throw new Error(`HTTP ${response.status}`);
      return response.json() as Promise<GuideMap>;
    })
    .catch((error: unknown) => {
      console.error('Guide map failed to load:', error);
      return {};
    })
    .then((map) => (guideMap = map));
  return guideMapPromise;
}

function loadFile(file: string): Promise<Record<string, Programme[]>> {
  const cached = schedules.get(file);
  if (cached) return cached;
  const pending = new Promise<Record<string, Programme[]>>((resolve, reject) => {
    worker ??= new Worker(new URL('./guide.worker.ts', import.meta.url), { type: 'module' });
    const activeWorker = worker;
    const onMessage = (event: MessageEvent<GuideResponse>) => {
      if (event.data.file !== file) return;
      activeWorker.removeEventListener('message', onMessage);
      if (event.data.ok) resolve(event.data.channels);
      else reject(new Error(event.data.error));
    };
    activeWorker.addEventListener('message', onMessage);
    activeWorker.postMessage({ file } satisfies GuideRequest);
  });
  pending.catch((error: unknown) => console.error(`Guide file ${file} failed:`, error));
  schedules.set(file, pending);
  return pending;
}

/** What the panel should show before any async work: settled answers when the map is already known. */
function initialState(stream: Stream): GuideState {
  if (typeof Worker === 'undefined' || !stream.channel) return { status: 'none' };
  if (guideMap && !guideEntry(guideMap, stream.id)) return { status: 'none' };
  return { status: 'loading' };
}

export function useGuide(stream: Stream): GuideState {
  const [state, setState] = useState<{ id: string; value: GuideState }>(() => ({
    id: stream.id,
    value: initialState(stream),
  }));

  useEffect(() => {
    let cancelled = false;
    const settle = (value: GuideState) => {
      if (!cancelled) setState({ id: stream.id, value });
    };
    const initial = initialState(stream);
    settle(initial);
    if (initial.status === 'none') return;

    void loadGuideMap().then((map) => {
      const entry = guideEntry(map, stream.id);
      if (!entry) return settle({ status: 'none' });
      return loadFile(entry.file).then(
        (channels) => {
          const programmes = channels[entry.channelId] ?? [];
          settle(programmes.length ? { status: 'ready', programmes } : { status: 'unavailable' });
        },
        () => settle({ status: 'unavailable' }),
      );
    });
    return () => {
      cancelled = true;
    };
    // Only the identity of the channel matters; the rest of the stream object is derived from it.
  }, [stream.id, stream.channel]);

  return state.id === stream.id ? state.value : initialState(stream);
}
```

import { useEffect, useState } from 'react';
import type { Stream } from '../types';
import type { Programme } from '../guide/parseXmltv';
import { useGuide } from '../guide/useGuide';

const UP_NEXT = 4;
const TICK_MS = 30_000;
const clock = new Intl.DateTimeFormat(undefined, { hour: '2-digit', minute: '2-digit' });

function useNow(interval: number) {
  const [now, setNow] = useState(() => Date.now());
  useEffect(() => {
    const timer = window.setInterval(() => setNow(Date.now()), interval);
    return () => window.clearInterval(timer);
  }, [interval]);
  return now;
}

function timeLeft(ms: number) {
  const minutes = Math.max(1, Math.round(ms / 60_000));
  if (minutes < 60) return `${minutes} min left`;
  const hours = Math.floor(minutes / 60);
  const rest = minutes % 60;
  return rest ? `${hours} h ${rest} min left` : `${hours} h left`;
}

const panel = 'mt-4 rounded-xl border border-line bg-panel p-4 md:p-5';
const label = 'font-mono text-[10px] tracking-[0.18em] text-dim uppercase';

/** Now/next schedule for channels that have guide data; renders nothing for channels that don't. */
export function GuidePanel({ stream }: { stream: Stream }) {
  const guide = useGuide(stream);
  const now = useNow(TICK_MS);

  if (guide.status === 'none') return null;

  if (guide.status === 'loading') {
    return (
      <div className={panel} aria-busy="true" aria-label="Loading guide">
        <div className="h-3 w-16 animate-pulse rounded bg-raised" />
        <div className="mt-3 h-6 w-2/3 animate-pulse rounded bg-raised" />
      </div>
    );
  }

  const current =
    guide.status === 'ready' ? guide.programmes.find((p) => p.start <= now && now < p.stop) : undefined;
  const upcoming =
    guide.status === 'ready' ? guide.programmes.filter((p) => p.start >= now && p !== current).slice(0, UP_NEXT) : [];

  if (!current && upcoming.length === 0) {
    return (
      <div className={panel}>
        <p className={label}>Guide unavailable</p>
      </div>
    );
  }

  return (
    <section aria-label="TV guide" className={`${panel} grid gap-5 md:grid-cols-[1.3fr_1fr] md:gap-8`}>
      {current && <OnNow programme={current} now={now} />}
      {upcoming.length > 0 && (
        <div className="min-w-0">
          <p className={label}>Up next</p>
          <ol className="mt-2 space-y-1.5">
            {upcoming.map((p) => (
              <li key={`${p.start}-${p.title}`} className="flex min-w-0 items-baseline gap-3 text-sm">
                <time dateTime={new Date(p.start).toISOString()} className="shrink-0 font-mono text-[12px] text-amber">
                  {clock.format(p.start)}
                </time>
                <span className="truncate text-paper">{p.title}</span>
              </li>
            ))}
          </ol>
        </div>
      )}
    </section>
  );
}

function OnNow({ programme, now }: { programme: Programme; now: number }) {
  const progress = Math.min(1, Math.max(0, (now - programme.start) / (programme.stop - programme.start)));
  return (
    <div className="min-w-0">
      <p className={label}>On now</p>
      <p className="mt-1.5 line-clamp-2 font-display text-2xl leading-tight font-extrabold tracking-tight uppercase md:text-3xl">
        {programme.title}
      </p>
      {programme.subtitle && <p className="mt-0.5 truncate text-sm text-paper/80">{programme.subtitle}</p>}
      <div className="mt-3 flex items-center gap-3">
        <span className="shrink-0 font-mono text-[12px] text-dim">
          {clock.format(programme.start)}–{clock.format(programme.stop)}
        </span>
        <div
          className="h-1 flex-1 overflow-hidden rounded-full bg-raised"
          role="progressbar"
          aria-label="Programme progress"
          aria-valuemin={0}
          aria-valuemax={100}
          aria-valuenow={Math.round(progress * 100)}
        >
          <div className="h-full rounded-full bg-amber transition-[width] duration-700" style={{ width: `${progress * 100}%` }} />
        </div>
        <span className="shrink-0 font-mono text-[12px] text-dim">{timeLeft(programme.stop - now)}</span>
      </div>
      {programme.description && <p className="mt-3 line-clamp-2 text-sm text-dim">{programme.description}</p>}
    </div>
  );
}
